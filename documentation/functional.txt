-- Create Intermediate Table (sales_orders_comp_stg)
CREATE TABLE sales_orders_comp_stg AS
SELECT 
  CONCAT(a.referenceto, a.refno) AS CompOrderNumber,
  a.stndard AS CompOrderType,
  b.bdcob AS CompSalesOrgCompanyCode,
  ...
FROM 
  (SELECT * FROM b_source.ord_hdr WHERE delete_flag <> 'D') a
  LEFT JOIN b_source.ord_dtl b 
    ON TRIM(a.referenceto) = TRIM(b.referenceto) 
    AND TRIM(a.refno) = TRIM(b.refno) 
    AND delete_flag <> 'D';

-- Create Main Table (sales_orders_comp)
CREATE TABLE sales_orders_comp AS
SELECT 
  a.OrderNumber AS CompOrderNumber,
  a.OrderType AS CompOrderType,
  a.SalesOrgCompanyCode AS CompSalesOrgCompanyCode,
  ...
FROM 
  s_master.sale_orders a
  LEFT JOIN b_source.ord_site e 
    ON TRIM(a.OrderNumber) = TRIM(CONCAT(e.referenceto, e.referencefrom)) 
    ...
  LEFT JOIN s_master.currency cur 
    ON TRIM(a.CurrencyType) = TRIM(cur.FromCurrency);
